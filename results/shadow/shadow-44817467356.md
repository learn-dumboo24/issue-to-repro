from click.shell_completion import split_arg_string, PowerShellComplete
from click.testing import CliRunner
import os

def test_split_arg_string_backslash():
    # POSIX shlex.split treats backslash as escape.
    # Current behavior with posix=True:
    # "C:\Users\ann\doc" -> "C:Usersanndoc" (if backslash followed by a char)
    
    # Based on the issue report:
    # print(split_arg_string(r"mycli sub C:\Users\ann\doc"))
    # Currently outputs: ['mycli', 'sub', 'C:Usersanndoc']
    
    # Expected:
    # ['mycli', 'sub', 'C:\\Users\\ann\\doc']
    
    result = split_arg_string(r"mycli sub C:\Users\ann\doc")
    assert result == ['mycli', 'sub', 'C:\\Users\\ann\\doc']

def test_powershell_completion_prog_name_with_space():
    # The second part of the issue:
    # The generated script has a second problem: source_vars() interpolates
    # prog_name into -CommandName %(prog_name)s and & %(prog_name)s without
    # quoting, so an executable path that contains a space produces a script that does
    # not parse.
    
    pc = PowerShellComplete(cli=None, ctx_args={}, prog_name="my cli", complete_var="_X")
    script = pc.source()
    
    # The template at line 214:
    # Register-ArgumentCompleter -Native -CommandName %(prog_name)s -ScriptBlock {
    # If prog_name is "my cli", it becomes -CommandName my cli, which PowerShell
    # parses as -CommandName my with an extra argument cli.
    
    # It should be '-CommandName "my cli"' or similar.
    
    assert "-CommandName 'my cli'" in script or '-CommandName "my cli"' in script
